const header = document.getElementById('site-header'); const menuToggle = document.getElementById('menu-toggle'); const mobileMenu = document.getElementById('mobile-menu'); const menuLines = document.querySelectorAll('.menu-line'); const mobileLinks = document.querySelectorAll('.mobile-link'); let menuOpen = false; /* Navbar scroll state */ window.addEventListener('scroll', () => { if (window.scrollY > 40) { header.classList.add( 'bg-kavora-green/95', 'backdrop-blur-xl', 'shadow-lg', 'shadow-black/10' ); header.querySelector('nav').classList.remove('h-24'); header.querySelector('nav').classList.add('h-20'); } else { header.classList.remove( 'bg-kavora-green/95', 'backdrop-blur-xl', 'shadow-lg', 'shadow-black/10' ); header.querySelector('nav').classList.remove('h-20'); header.querySelector('nav').classList.add('h-24'); } }); /* Mobile menu */ function toggleMenu() { menuOpen = !menuOpen; menuToggle.setAttribute('aria-expanded', menuOpen); if (menuOpen) { mobileMenu.classList.remove( 'pointer-events-none', 'invisible', 'opacity-0' ); mobileMenu.classList.add( 'pointer-events-auto', 'visible', 'opacity-100' ); menuLines[0].classList.add('rotate-45', 'top-1/2'); menuLines[1].classList.add('opacity-0'); menuLines[2].classList.add('-rotate-45', 'bottom-auto', 'top-1/2'); } else { mobileMenu.classList.remove( 'pointer-events-auto', 'visible', 'opacity-100' ); mobileMenu.classList.add( 'pointer-events-none', 'invisible', 'opacity-0' ); menuLines[0].classList.remove('rotate-45', 'top-1/2'); menuLines[1].classList.remove('opacity-0'); menuLines[2].classList.remove('-rotate-45', 'bottom-auto', 'top-1/2'); } } menuToggle.addEventListener('click', toggleMenu); /* Close mobile menu when a link is clicked */ mobileLinks.forEach(link => { link.addEventListener('click', () => { if (menuOpen) { toggleMenu(); } }); });
